// ============ STATE ============
let state = {
  currentUnit: 0,
  currentLesson: 0,
  currentTab: 'lesson',
  completedLessons: [],
  quizScores: {},
  xp: 0,
  level: 1,
  theme: 'light',
  role: 'student',
  userName: 'Student',
  achievements: [],
  streak: 0,
  lastActive: null,
  quizAttempts: {}
};

// ============ INIT ============
function init() {
  loadState();
  renderSyllabus();
  renderLesson();
  updateProgress();
  updateXP();
  setupThemeButtons();
  setupTabs();
  setupRoleSwitcher();
}

function loadState() {
  const saved = localStorage.getItem('scienceHubState');
  if (saved) {
    state = { ...state, ...JSON.parse(saved) };
    document.body.setAttribute('data-theme', state.theme);
    document.getElementById('userName').textContent = state.userName;
    document.getElementById('roleSwitcher').value = state.role;
    updateThemeButtons();
  }
}

function saveState() {
  localStorage.setItem('scienceHubState', JSON.stringify(state));
}

// ============ SYLLABUS RENDERING ============
function renderSyllabus() {
  const list = document.getElementById('syllabusList');
  list.innerHTML = '';
  syllabus.forEach((unit, ui) => {
    const unitDiv = document.createElement('div');
    unitDiv.className = 'mb-2';
    unitDiv.innerHTML = `
      <div class="sidebar-item ${state.currentUnit === ui ? 'active' : ''}" onclick="selectUnit(${ui})" style="${state.currentUnit === ui ? `background: ${unit.color}` : ''}">
        <span class="text-xl">${unit.icon}</span>
        <div class="flex-1">
          <div class="font-semibold text-sm">${unit.title}</div>
          <div class="text-xs opacity-75">${unit.lessons.length} lessons</div>
        </div>
      </div>
      <div class="ml-6 space-y-1 ${state.currentUnit === ui ? '' : 'hidden'}" id="unit-${ui}-lessons">
        ${unit.lessons.map((lesson, li) => {
          const isComplete = state.completedLessons.includes(`${unit.id}.${lesson.id}`);
          const isActive = state.currentUnit === ui && state.currentLesson === li;
          return `
            <div class="sidebar-item text-sm ${isActive ? 'active' : ''}" onclick="selectLesson(${ui}, ${li})" style="${isActive ? `background: ${unit.color}` : ''}">
              <span>${isComplete ? '✅' : '📄'}</span>
              <span class="flex-1">${lesson.title}</span>
            </div>
          `;
        }).join('')}
      </div>
    `;
    list.appendChild(unitDiv);
  });
  document.getElementById('totalLessons').textContent = syllabus.reduce((sum, u) => sum + u.lessons.length, 0);
}

function selectUnit(ui) {
  state.currentUnit = ui;
  state.currentLesson = 0;
  renderSyllabus();
  renderLesson();
  saveState();
}

function selectLesson(ui, li) {
  state.currentUnit = ui;
  state.currentLesson = li;
  state.currentTab = 'lesson';
  renderSyllabus();
  renderLesson();
  updateTabButtons();
  saveState();
}

function navigateLesson(dir) {
  const unit = syllabus[state.currentUnit];
  const newLesson = state.currentLesson + dir;
  if (newLesson >= 0 && newLesson < unit.lessons.length) {
    state.currentLesson = newLesson;
  } else if (dir > 0 && state.currentUnit < syllabus.length - 1) {
    state.currentUnit++;
    state.currentLesson = 0;
  } else if (dir < 0 && state.currentUnit > 0) {
    state.currentUnit--;
    state.currentLesson = syllabus[state.currentUnit].lessons.length - 1;
  }
  renderSyllabus();
  renderLesson();
  saveState();
}

// ============ LESSON RENDERING ============
function renderLesson() {
  const unit = syllabus[state.currentUnit];
  const lesson = unit.lessons[state.currentLesson];
  const content = document.getElementById('mainContent');

  if (state.currentTab === 'lesson') {
    content.innerHTML = lesson.content;
  } else if (state.currentTab === 'keywords') {
    content.innerHTML = `
      <h2 class="text-2xl font-bold mb-4" style="color: var(--primary);">🔑 Keywords: ${lesson.title}</h2>
      <p class="mb-4" style="color: var(--text-light);">Master these keywords to understand this lesson better. Click each to hear pronunciation (simulated).</p>
      <div class="flex flex-wrap gap-2 mb-6">
        ${lesson.keywords.map(kw => `
          <button class="keyword-tag hover:scale-110 transition-transform" onclick="speakKeyword('${kw}')">
            🔊 ${kw}
          </button>
        `).join('')}
      </div>
      <div class="card p-4 mt-4">
        <h3 class="font-bold mb-2">📝 Practice</h3>
        <p class="text-sm mb-3">Write each keyword in a sentence:</p>
        <div class="space-y-2">
          ${lesson.keywords.map((kw, i) => `
            <div class="flex gap-2 items-center">
              <span class="font-semibold text-sm w-32" style="color: var(--primary);">${kw}:</span>
              <input type="text" class="flex-1 px-3 py-2 rounded border-2 text-sm" style="border-color: var(--border); background: var(--bg); color: var(--text);" placeholder="Write a sentence with '${kw}'..." id="kw-${i}">
            </div>
          `).join('')}
        </div>
        <button class="btn-primary mt-4" onclick="saveKeywords()">💾 Save my sentences</button>
      </div>
    `;
  } else if (state.currentTab === 'quiz') {
    renderQuiz(lesson);
  } else if (state.currentTab === 'activity') {
    content.innerHTML = `
      <h2 class="text-2xl font-bold mb-4" style="color: var(--primary);">🧪 Activity: ${lesson.title}</h2>
      <div class="card p-6 bg-gradient-to-br from-blue-50 to-purple-50 border-2 border-dashed" style="border-color: var(--primary);">
        <div class="text-4xl mb-3">🔬</div>
        <h3 class="font-bold text-lg mb-3">Instructions</h3>
        <p class="mb-4">${lesson.activity}</p>
        <div class="bg-white p-4 rounded-lg">
          <h4 class="font-bold mb-2">📋 My observations:</h4>
          <textarea class="w-full p-3 border-2 rounded min-h-[120px]" style="border-color: var(--border); background: var(--bg); color: var(--text);" placeholder="Write what you observed..." id="activityNotes"></textarea>
        </div>
        <button class="btn-primary mt-4" onclick="saveActivity()">💾 Save observations</button>
      </div>
    `;
  }

  // Update nav buttons
  document.getElementById('prevBtn').disabled = (state.currentUnit === 0 && state.currentLesson === 0);
  const isLastLesson = state.currentUnit === syllabus.length - 1 && state.currentLesson === syllabus[syllabus.length-1].lessons.length - 1;
  document.getElementById('nextBtn').disabled = isLastLesson;

  const lessonId = `${unit.id}.${lesson.id}`;
  const isComplete = state.completedLessons.includes(lessonId);
  document.getElementById('markCompleteBtn').textContent = isComplete ? '✅ Completed' : '✓ Mark Complete';
  document.getElementById('markCompleteBtn').disabled = isComplete;
}

// ============ QUIZ ============
function renderQuiz(lesson) {
  const content = document.getElementById('mainContent');
  const lessonId = `${syllabus[state.currentUnit].id}.${lesson.id}`;
  const attempts = state.quizAttempts[lessonId] || 0;
  const bestScore = state.quizScores[lessonId] || 0;

  content.innerHTML = `
    <h2 class="text-2xl font-bold mb-2" style="color: var(--primary);"> Quiz: ${lesson.title}</h2>
    <div class="flex gap-4 mb-4 flex-wrap">
      <span class="badge" style="background: var(--primary); color: white;">Attempts: ${attempts}</span>
      <span class="badge" style="background: var(--accent); color: white;">Best: ${bestScore}/${lesson.quiz.length}</span>
    </div>
    <div id="quizContainer">
      ${lesson.quiz.map((q, qi) => `
        <div class="card p-4 mb-4" id="quiz-q-${qi}">
          <p class="font-bold mb-3">${qi + 1}. ${q.q}</p>
          <div class="space-y-2">
            ${q.options.map((opt, oi) => `
              <div class="quiz-option" onclick="selectAnswer(${qi}, ${oi})" id="q${qi}-o${oi}">
                ${String.fromCharCode(65 + oi)}. ${opt}
              </div>
            `).join('')}
          </div>
        </div>
      `).join('')}
      <button class="btn-primary w-full py-3 text-lg" onclick="submitQuiz()">📊 Submit Answers</button>
    </div>
  `;
}

let selectedAnswers = {};

function selectAnswer(qi, oi) {
  selectedAnswers[qi] = oi;
  const lesson = syllabus[state.currentUnit].lessons[state.currentLesson];
  lesson.quiz[qi].options.forEach((_, i) => {
    document.getElementById(`q${qi}-o${i}`).classList.remove('selected');
  });
  document.getElementById(`q${qi}-o${oi}`).classList.add('selected');
}

function submitQuiz() {
  const lesson = syllabus[state.currentUnit].lessons[state.currentLesson];
  const lessonId = `${syllabus[state.currentUnit].id}.${lesson.id}`;
  let correct = 0;

  lesson.quiz.forEach((q, qi) => {
    const selected = selectedAnswers[qi];
    if (selected === undefined) {
      alert(`Please answer question ${qi + 1}`);
      return;
    }
    const el = document.getElementById(`q${qi}-o${selected}`);
    if (selected === q.correct) {
      el.classList.add('correct');
      correct++;
    } else {
      el.classList.add('wrong');
      document.getElementById(`q${qi}-o${q.correct}`).classList.add('correct');
    }
    // Disable all options
    q.options.forEach((_, oi) => {
      document.getElementById(`q${qi}-o${oi}`).style.pointerEvents = 'none';
    });
  });

  const attempts = (state.quizAttempts[lessonId] || 0) + 1;
  state.quizAttempts[lessonId] = attempts;
  const prevBest = state.quizScores[lessonId] || 0;
  if (correct > prevBest) {
    state.quizScores[lessonId] = correct;
  }

  const xpEarned = correct * 10;
  addXP(xpEarned);

  setTimeout(() => {
    showAchievement(
      correct === lesson.quiz.length ? '🏆' : correct >= lesson.quiz.length / 2 ? '⭐' : '📚',
      correct === lesson.quiz.length ? 'Perfect Score!' : correct >= lesson.quiz.length / 2 ? 'Good Job!' : 'Keep Trying!',
      `You got ${correct}/${lesson.quiz.length} correct! +${xpEarned} XP`,
      `+${xpEarned} XP`
    );
  }, 1000);

  saveState();
  updateProgress();
}

// ============ TABS ============
function setupTabs() {
  document.querySelectorAll('.tab-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      state.currentTab = btn.dataset.tab;
      updateTabButtons();
      renderLesson();
      saveState();
    });
  });
}

function updateTabButtons() {
  document.querySelectorAll('.tab-btn').forEach(btn => {
    btn.classList.toggle('active', btn.dataset.tab === state.currentTab);
  });
}

// ============ THEME ============
function setupThemeButtons() {
  document.querySelectorAll('.theme-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      const theme = btn.dataset.themeSet;
      state.theme = theme;
      document.body.setAttribute('data-theme', theme);
      updateThemeButtons();
      saveState();
    });
  });
}

function updateThemeButtons() {
  document.querySelectorAll('.theme-btn').forEach(btn => {
    btn.classList.toggle('active', btn.dataset.themeSet === state.theme);
  });
}

// ============ ROLE ============
function setupRoleSwitcher() {
  document.getElementById('roleSwitcher').addEventListener('change', (e) => {
    state.role = e.target.value;
    saveState();
    if (state.role !== 'student') {
      showReport();
    }
  });
}

// ============ PROGRESS ============
function markLessonComplete() {
  const unit = syllabus[state.currentUnit];
  const lesson = unit.lessons[state.currentLesson];
  const lessonId = `${unit.id}.${lesson.id}`;

  if (!state.completedLessons.includes(lessonId)) {
    state.completedLessons.push(lessonId);
    addXP(25);
    showAchievement('✅', 'Lesson Complete!', `You finished ${lesson.title}`, '+25 XP');
    updateProgress();
    renderSyllabus();
    renderLesson();
    saveState();
  }
}

function addXP(amount) {
  state.xp += amount;
  const newLevel = Math.floor(state.xp / 100) + 1;
  if (newLevel > state.level) {
    state.level = newLevel;
    setTimeout(() => {
      showAchievement('🎉', 'Level Up!', `You reached Level ${newLevel}!`, `Level ${newLevel}`);
    }, 1500);
  }
  updateXP();
  saveState();
}

function updateXP() {
  document.getElementById('userXP').textContent = state.xp;
  document.getElementById('userLevel').textContent = `Level ${state.level}`;
  const xpInLevel = state.xp % 100;
  document.getElementById('xpBar').style.width = `${xpInLevel}%`;
}

function updateProgress() {
  const total = syllabus.reduce((sum, u) => sum + u.lessons.length, 0);
  const completed = state.completedLessons.length;
  const pct = (completed / total) * 100;
  document.getElementById('totalProgress').style.width = `${pct}%`;
  document.getElementById('completedLessons').textContent = completed;
}

function renderTeacherChart() {
  const ctx = document.getElementById('teacherChart');
  if (!ctx) return;
  new Chart(ctx, {
    type: 'bar',
    data: {
      labels: syllabus.map(u => u.title),
      datasets: [{
        label: 'Completion %',
        data: syllabus.map(unit => {
          const done = state.completedLessons.filter(id => id.startsWith(unit.id + '.')).length;
          return Math.round((done / unit.lessons.length) * 100);
        }),
        backgroundColor: syllabus.map(u => u.color)
      }]
    },
    options: { responsive: true, maintainAspectRatio: false, scales: { y: { beginAtZero: true, max: 100 } } }
  });
}

function renderParentChart() {
  const ctx = document.getElementById('parentChart');
  if (!ctx) return;
  new Chart(ctx, {
    type: 'radar',
    data: {
      labels: syllabus.map(u => u.title.replace(/\d+\.\d+\s*/, '')),
      datasets: [{
        label: 'Progress',
        data: syllabus.map(unit => {

}

function getMotivationalMessage(pct) {
  if (pct === 100) return "🌟 Amazing! You're a Science Superstar!";
  if (pct >= 80) return "🎉 Excellent work! Keep it up!";
  if (pct >= 60) return "👍 Great progress! You're doing well!";
  if (pct >= 40) return "💪 Good effort! Keep going!";
  if (pct >= 20) return "📚 Nice start! Every lesson counts!";
  return " Let's begin this science journey!";
}

function getProgressSummary(pct) {
  if (pct >= 80) return `${state.userName} is doing excellent work and shows strong understanding of the material.`;
  if (pct >= 50) return `${state.userName} is making good progress. With continued effort, they will master all topics.`;
  if (pct >= 20) return `${state.userName} has started their learning journey. Regular practice will help them improve.`;
  return `${state.userName} is at the beginning of their learning. Encouragement and support will help them engage more.`;
}

function getStrengths() {
  const strengths = [];
  syllabus.forEach(unit => {
    const done = state.completedLessons.filter(id => id.startsWith(unit.id + '.')).length;
    if (done === unit.lessons.length) strengths.push(`Completed all ${unit.title} lessons`);
  });
  const quizKeys = Object.keys(state.quizScores);
  if (quizKeys.length > 0) {
    const avg = getAverageQuizScore();
    if (avg >= 80) strengths.push('Excellent quiz performance');
    if (avg >= 60) strengths.push('Good quiz understanding');
  }
  if (state.xp >= 200) strengths.push('High engagement (XP earned)');
  if (strengths.length === 0) strengths.push('Enthusiasm to learn', 'Willingness to try');
  return strengths;
}

function getImprovements() {
  const improvements = [];
  syllabus.forEach(unit => {
    const done = state.completedLessons.filter(id => id.startsWith(unit.id + '.')).length;
    if (done < unit.lessons.length) improvements.push(`Complete remaining ${unit.title} lessons`);
  });
  const quizKeys = Object.keys(state.quizScores);
  if (quizKeys.length > 0) {
    const avg = getAverageQuizScore();
    if (avg < 70) improvements.push('Review quiz topics for better understanding');
  }
  if (improvements.length === 0) improvements.push('Challenge with advanced topics', 'Help peers learn'];
  return improvements;
}

function getTeacherNote(pct) {
  if (pct >= 80) return `${state.userName} demonstrates excellent understanding and consistent effort. They are ready for more challenging material.`;
  if (pct >= 50) return `${state.userName} is progressing well. Encourage them to complete remaining lessons and review quiz topics.`;
  if (pct >= 20) return `${state.userName} has started engaging with the material. Provide additional support and encouragement.`;
  return `${state.userName} needs more engagement. Consider interactive activities and positive reinforcement.`;
}

function getAverageQuizScore() {
  const keys = Object.keys(state.quizScores);
  if (keys.length === 0) return 0;
  let total = 0;
  keys.forEach(k => {
    const unitId = k.split('.')[0];
    const lessonId = k.split('.')[1];
    const unit = syllabus.find(u => u.id == unitId);
    if (unit) {
      const lesson = unit.lessons.find(l => l.id === lessonId);
      if (lesson) total += (state.quizScores[k] / lesson.quiz.length) * 100;
    }
  });
  return Math.round(total / keys.length);
}

}

function speakKeyword(kw) {

    const utterance = new SpeechSynthesisUtterance(kw);
    utterance.rate = 0.8;
    speechSynthesis.speak(utterance);
  }
}

function saveKeywords() {
  const unit = syllabus[state.currentUnit];
  const lesson = unit.lessons[state.currentLesson];
  const sentences = lesson.keywords.map((_, i) => document.getElementById(`kw-${i}`).value).filter(v => v);
  if (sentences.length > 0) {
    addXP(10);
    showAchievement('', 'Keywords Saved!', `You wrote ${sentences.length} sentences`, '+10 XP');
  }
}

  }
}

function resetProgress() {
